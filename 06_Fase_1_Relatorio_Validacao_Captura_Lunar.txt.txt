4.4 — Lua


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  r_lua_AU  : 80 × RE_AU
            = 80 × (6.371×10⁶ / 1.496×10¹¹)
            = 80 × 4.259×10⁻⁵ AU
            = 3.407×10⁻³ AU
            = 80 R⊕  ✅
  ORIGEM DO 80 R⊕:
  → Recuo lunar por maré = 3.8 cm/ano
  → Extrapolação retroativa via dados
    geofísicos (Touma & Wisdom 1994)
  → Consistente com Rhythmites de
    Elatina, Austrália (~620 Ma)
  pos_x : r_terra_0 + r_lua_AU
        = 0.990 + 3.407×10⁻³ AU
  pos_y : 0.0
  v_lua_rel : v_circ(r_lua_AU, M_terra) × 0.85
            = √(4π² × M_terra / r_lua_AU) × 0.85
  ORIGEM DO 0.85:
  → v < v_escape → energia total < 0
  → corpo em regime de órbita ligada
  → fator derivado por conservação
    de energia considerando Marte
    como perturbador na res. 3:2
  v_lua_abs : v_terra + v_lua_rel
            → velocidade no referencial
              inercial do Sol  ✅
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
