import rebound
import numpy as np
from scipy.optimize import differential_evolution

AU_km        = 1.496e8
m_marte      = 3.213e-7 * 0.95          # ← 95% da massa atual de Marte
a_marte_i    = 1.3572
a_marte_alvo = 1.5237
contador = [0]
melhor   = [999.0]

def simular(params):
    (a1, e1, inc1, M1, m1,
     a2, e2, inc2, M2,
     M_marte_deg) = params
    contador[0] += 1

    try:
        sim = rebound.Simulation()
        sim.units = ('AU', 'yr', 'Msun')
        sim.integrator = "ias15"
        sim.ri_ias15.epsilon = 1e-7
        sim.dt = 0.05

        sim.add(m=1.0)
        sim.add(m=m_marte,
                a=a_marte_i, e=0.08,
                inc=np.radians(3.0),
                M=np.radians(M_marte_deg))
        sim.add(m=m1,
                a=a1, e=e1,
                inc=np.radians(inc1),
                M=np.radians(M1))
        sim.add(m=5e-8,
                a=a2, e=e2,
                inc=np.radians(inc2),
                M=np.radians(M2))
        sim.add(m=9.548e-4,
                a=5.2, e=0.049,
                inc=np.radians(1.3),
                M=np.radians(0.0))
        sim.move_to_com()

        a_marte_track = []
        for t in np.linspace(0, 500, 100):
            sim.integrate(t)
            a_track = sim.particles[1].a
            if a_track <= 0 or a_track > 8:
                return 999.0
            a_marte_track.append(a_track)

        a_final = np.mean(a_marte_track[-20:])
        delta_a = abs(a_final - a_marte_alvo)

        if delta_a < melhor[0]:
            melhor[0] = delta_a
            print(f"  🔥 [{contador[0]}] a_Marte={a_final:.5f} AU "
                  f"Δ={delta_a:.5f} | "
                  f"m1={m1:.2e} a1={a1:.2f} e1={e1:.2f} "
                  f"a2={a2:.2f} e2={e2:.2f}")
        elif contador[0] % 50 == 0:
            print(f"  [{contador[0]}] melhor Δa={melhor[0]:.5f} AU")

        return delta_a

    except:
        return 999.0

print("🔍 Abordagem B+A RÁPIDA — Sistema caótico pós-captura")
print(f"   Marte: m={m_marte:.3e} | a_i={a_marte_i} → alvo={a_marte_alvo} AU")
print(f"   ~30 min\n")

bounds = [
    (1.5,  4.0),    # a1  proto-planeta
    (0.1,  0.85),   # e1
    (0.0,  15.0),   # inc1
    (0.0,  360.0),  # M1
    (1e-5, 5e-4),   # m1  ← massa livre
    (1.3,  3.5),    # a2  asteroide
    (0.1,  0.90),   # e2
    (0.0,  10.0),   # inc2
    (0.0,  360.0),  # M2
    (0.0,  360.0),  # M_marte
]

resultado = differential_evolution(
    simular,
    bounds,
    maxiter=20,
    popsize=5,
    tol=1e-6,
    seed=42,
    mutation=(0.6, 1.8),
    recombination=0.85,
    polish=True,
    disp=True,
    workers=1,
)

p = resultado.x
delta_final = resultado.fun

# ── recalcula final ──────────────────────────────
sim2 = rebound.Simulation()
sim2.units = ('AU', 'yr', 'Msun')
sim2.integrator = "ias15"
sim2.add(m=1.0)
sim2.add(m=m_marte, a=a_marte_i, e=0.08,
         inc=np.radians(3.0), M=np.radians(p[9]))
sim2.add(m=p[4], a=p[0], e=p[1],
         inc=np.radians(p[2]), M=np.radians(p[3]))
sim2.add(m=5e-8, a=p[5], e=p[6],
         inc=np.radians(p[7]), M=np.radians(p[8]))
sim2.add(m=9.548e-4, a=5.2, e=0.049,
         inc=np.radians(1.3), M=np.radians(0.0))
sim2.move_to_com()

track = []
for t in np.linspace(0, 500, 100):
    sim2.integrate(t)
    track.append(sim2.particles[1].a)

a_final = np.mean(track[-20:])

print(f"\n{'='*55}")
print(f"🏆 RESULTADO FINAL:")
print(f"   Proto-planeta:")
print(f"     a1 = {p[0]:.4f} AU | e1  = {p[1]:.4f}")
print(f"     inc= {p[2]:.4f}°  | M1  = {p[3]:.4f}°")
print(f"     m1 = {p[4]:.4e} Msun ({p[4]/3.213e-7:.1f}x Marte)")
print(f"   Asteroide:")
print(f"     a2 = {p[5]:.4f} AU | e2  = {p[6]:.4f}")
print(f"     inc= {p[7]:.4f}°  | M2  = {p[8]:.4f}°")
print(f"   M_marte = {p[9]:.4f}°")
print(f"\n   a_Marte inicial = {a_marte_i} AU")
print(f"   a_Marte final   = {a_final:.5f} AU")
print(f"   a_Marte alvo    = {a_marte_alvo} AU")
print(f"   Δa residual     = {delta_final:.6f} AU")

if delta_final < 0.005:
    print("\n   ✅ DESLOCAMENTO CONFIRMADO!")
elif delta_final < 0.02:
    print("\n   🔥 MUITO PRÓXIMO — zoom fino!")
elif delta_final < 0.05:
    print("\n   ⚡ PROGRESSO REAL — refinar!")
else:
    print(f"\n   ⚠️  Δa={delta_final:.4f} — aumentar massa ou tempo")
